async function updateData() { try { const res = await fetch('https://docs.google.com/spreadsheets/d/1IMXl6rCvbTRFNfeypfY4KpEdeFAsnJ0rh04t46ujcoI/export?format=csv'); const csvText = await res.text(); const rows = csvText.split('\n'); const columns = rows[0].split(','); const niftyPrice = columns[0].replace(/"/g, '').trim(); // Column E (Index 4) se AI Signal uthana const aiSignal = columns[4].replace(/"/g, '').trim(); if(niftyPrice) { document.getElementById('nifty').innerText = "₹" + niftyPrice; } if(aiSignal) { document.getElementById('signal').innerText = aiSignal; // Dynamic Colors if(aiSignal.includes("BUY")) { document.getElementById('signal').style.color = "#4ade80"; } else if(aiSignal.includes("SELL")) { document.getElementById('signal').style.color = "#f87171"; } else { document.getElementById('signal').style.color = "#f1c40f"; } } document.getElementById('time').innerText = new Date().toLocaleTimeString(); } catch (e) { console.log("Re-Syncing Indicators..."); } }
